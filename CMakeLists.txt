cmake_minimum_required(VERSION 3.22.1)
project(thashlib VERSION 0.1.1 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
option(BUILD_SHARED_LIBS "Build shared libraries" ON)
find_package(TclStub 8.6 REQUIRED)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)

add_library(thashlib src/library.c)
set_target_properties(thashlib PROPERTIES POSITION_INDEPENDENT_CODE ON PREFIX "lib")
target_include_directories(thashlib PRIVATE ${TCL_INCLUDE_PATH})
target_compile_definitions(thashlib PRIVATE BUILD_thashlib THASHLIB_VERSION="${PROJECT_VERSION}")
target_link_libraries(thashlib PRIVATE OpenSSL::Crypto)
if(BUILD_SHARED_LIBS)
    target_compile_definitions(thashlib PRIVATE USE_TCL_STUBS)
    target_link_libraries(thashlib PRIVATE ${TCL_STUB_LIBRARY})
    configure_file(pkgIndex.tcl.in pkgIndex.tcl @ONLY)
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl DESTINATION lib/thashlib${PROJECT_VERSION})
else()
    target_compile_definitions(thashlib PRIVATE STATIC_BUILD)
endif()
install(TARGETS thashlib LIBRARY DESTINATION lib/thashlib${PROJECT_VERSION}
    ARCHIVE DESTINATION lib/thashlib${PROJECT_VERSION} RUNTIME DESTINATION lib/thashlib${PROJECT_VERSION})
install(FILES README.md LICENSE DESTINATION share/doc/thashlib)

include(CTest)
if(BUILD_TESTING)
    if(BUILD_SHARED_LIBS)
        add_test(NAME thashlib COMMAND ${TCL_TCLSH} ${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl $<TARGET_FILE_DIR:thashlib>)
        add_test(NAME unavailable-provider COMMAND ${TCL_TCLSH} ${CMAKE_CURRENT_SOURCE_DIR}/tests/unavailable.tcl $<TARGET_FILE_DIR:thashlib>)
        # Multi-configuration generators put the DLL in a configuration subdirectory.
        file(GENERATE OUTPUT "$<TARGET_FILE_DIR:thashlib>/pkgIndex.tcl" INPUT "${CMAKE_CURRENT_BINARY_DIR}/pkgIndex.tcl")
    else()
        find_library(THASHLIB_TCL_LIBRARY NAMES tcl${TCL_VERSION} tcl${TCL_ABI_VERSION} tcl9${TCL_ABI_VERSION} tcl
            HINTS ${TCL_LIBRARY_DIR} "${TCL_INCLUDE_PATH_PARENT}/lib" REQUIRED)
        add_executable(thashlib-static-test tests/static.c)
        target_include_directories(thashlib-static-test PRIVATE ${TCL_INCLUDE_PATH} src)
        target_link_libraries(thashlib-static-test PRIVATE thashlib ${THASHLIB_TCL_LIBRARY})
        add_test(NAME thashlib COMMAND thashlib-static-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl)
        add_test(NAME unavailable-provider COMMAND thashlib-static-test ${CMAKE_CURRENT_SOURCE_DIR}/tests/unavailable.tcl)
    endif()
    set_tests_properties(unavailable-provider PROPERTIES ENVIRONMENT "OPENSSL_CONF=${CMAKE_CURRENT_SOURCE_DIR}/tests/unavailable.cnf")
endif()
